🧩 How to run my School Management System
# clone
git clone https://github.com/TrixieMisheen/school-management-system.git
cd school-management-system

# create venv
python -m venv .venv
source .venv/bin/activate   # or .venv\Scripts\activate on Windows

# install
pip install -r requirements.txt

# start backend
cd backend
flask run

# start frontend
python ui/app.py
